    static void Check()
    {
        if (!EditorApplication.isPlayingOrWillChangePlaymode && !EditorApplication.isCompiling && !EditorApplication.isUpdating && File.Exists(Folder + "/polish.request"))
        {
            File.Delete(Folder + "/polish.request");
            try
            {
                var game = SceneManager.GetSceneByName("GameScene");
                var session = game.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<BattleSession>(true)).Single();
                var visibility = game.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<GameplayHudVisibility>(true)).FirstOrDefault();
                if (visibility != null) session.explorationHUD = new[] { Field<GameObject>(visibility, "experienceHud"), Field<GameObject>(visibility, "minimap"), Field<GameObject>(visibility, "portraitHud") };
                foreach (var character in session.party)
                {
                    AddAnimations(character.GetComponentInChildren<Animator>(true));
                    character.GetComponent<BattleUnit>().FitBattleHitbox();
                }
                EditorSceneManager.MarkSceneDirty(game); EditorSceneManager.SaveScene(game);
                var battle = EditorSceneManager.OpenScene("Assets/Scenes/BattleScene.unity", OpenSceneMode.Additive);
                foreach (var volume in battle.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<Volume>())) volume.priority = 20;
                var controller = battle.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<BattleController>(true)).Single();
                controller.hud.GetComponent<CanvasScaler>().screenMatchMode = CanvasScaler.ScreenMatchMode.Expand;
                foreach (var skill in controller.skills)
                {
                    string prefix = new[] { "Knight", "Wizard", "Healer" }[skill.character];
                    string file = prefix + "_" + skill.displayName.Replace(" ", "");
                    skill.icon = SpriteAsset("Assets/Imported Assets/UI/" + prefix + "/" + file + ".png");
                    EditorUtility.SetDirty(skill);
                }
                foreach (var button in controller.hud.GetComponentsInChildren<Button>(true)) if (button.targetGraphic != null) button.targetGraphic.raycastTarget = true;
                var boss = controller.bossGame;
                boss.heart.GetComponent<Image>().sprite = SpriteAsset("Assets/Imported Assets/Jevil/Sprites/soul.png");
                boss.jevil.GetComponent<Image>().sprite = SpriteAsset("Assets/Imported Assets/Jevil/Sprites/jevil.png");
                boss.spadeTemplate.sprite = SpriteAsset("Assets/Imported Assets/Jevil/Bullets/Jevil Spade.png");
                boss.heartTemplate.sprite = SpriteAsset("Assets/Imported Assets/Jevil/Bullets/Jevil Heart.png");
                boss.scytheTemplate.sprite = SpriteAsset("Assets/Imported Assets/Jevil/Bullets/jevil_knife.png");
                controller.settings.music = Audio("Assets/Imported Assets/Audio/Music/09 Lace.mp3") ?? Audio("Assets/Imported Assets/Audio/Music/Smart Race.mp3");
                controller.settings.victorySound = FindAudio("positive");
                controller.settings.dragon = Dragon();
                controller.settings.laser = MinionProjectileSetup.BuildPrefab();
                controller.settings.areaEffect = AreaEffect(); controller.settings.statusEffect = StatusEffect();
                controller.settings.shieldSound = FindAudio("divine_shield"); controller.settings.portalSound = FindAudio("snd_dash_charge");
                controller.settings.teleportSound = FindAudio("teleport_poof"); controller.settings.laserSound = FindAudio("laser_fire");
                controller.settings.chargeSound = FindAudio("magic_charge"); controller.settings.melodySound = FindAudio("lullaby_tight");
                foreach (var layout in controller.layouts)
                {
                    foreach (var renderer in layout.transform.Find("Floor").GetComponentsInChildren<Renderer>(true))
                    {
                        var materials = renderer.sharedMaterials;
                        for (int i = 0; i < materials.Length; i++) materials[i] = Material("ArenaStone" + ((renderer.transform.GetSiblingIndex() + i) % 3), new Color(.43f + i * .025f, .47f + i * .025f, .5f + i * .025f));
                        renderer.sharedMaterials = materials;
                    }
                    foreach (var cover in layout.cover) cover.shape.GetComponent<Renderer>().sharedMaterial = Material("ArenaCover", new Color(.19f, .23f, .29f));
                }
                foreach (var data in controller.settings.enemies.Concat(new[] { controller.settings.boss }))
                {
                    string path = AssetDatabase.GetAssetPath(data.prefab); var prefab = PrefabUtility.LoadPrefabContents(path);
                    var highlight = new SerializedObject(prefab.GetComponent<ObjectHighlight>());
                    highlight.FindProperty("outlineWidth").floatValue = .003f; highlight.ApplyModifiedPropertiesWithoutUndo();
                    PrefabUtility.SaveAsPrefabAsset(prefab, path); PrefabUtility.UnloadPrefabContents(prefab);
                }
                EditorUtility.SetDirty(controller.settings); AssetDatabase.SaveAssets();
                EditorSceneManager.MarkSceneDirty(battle); EditorSceneManager.SaveScene(battle); EditorSceneManager.CloseScene(battle, true);
                File.WriteAllText(Folder + "/polish-result.txt", "PASS: exploration HUD suspended during battle; battle post-processing priority set.");
            }
            catch (Exception error) { File.WriteAllText(Folder + "/polish-result.txt", error.ToString()); }
        }
        if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating || !File.Exists(Folder + "/setup.request")) return;
        File.Delete(Folder + "/setup.request");
        try { Build(); File.WriteAllText(Folder + "/setup-result.txt", "PASS: BattleScene, three layouts, 12 skill assets, Skeleton/Minion/Jevil encounter prefabs, NavMesh surfaces, saved battle UI, transitions and scene wiring saved."); }
        catch (Exception error) { File.WriteAllText(Folder + "/setup-result.txt", error.ToString()); Debug.LogException(error); }
    }

    static GameObject Enemy(BattleEnemyData data)
    {
        var root = new GameObject(data.displayName); root.layer = 11;
        var visual = new GameObject("Visual"); visual.transform.SetParent(root.transform, false);
        var motion = root.AddComponent<BattleEnemyMotion>(); motion.visual = visual.transform;
        if (data.kind == BattleEnemyKind.Skeleton)
        {
            DungeonPolishSetup.AttachSkeleton(visual.transform);
        }
        else
        {
            var source = data.kind == BattleEnemyKind.Minion ? Load<GameObject>("Assets/Resources/Dungeon/FriendlyMinion.prefab") : Load<GameObject>("Assets/Imported Assets/Jevil/Model/Jevil.fbx");
            var model = Object.Instantiate(source, visual.transform);
            foreach (var canvas in model.GetComponentsInChildren<Canvas>(true)) if (canvas != null) Object.DestroyImmediate(canvas.gameObject);
            foreach (var behaviour in model.GetComponentsInChildren<MonoBehaviour>(true)) if (behaviour != null) Object.DestroyImmediate(behaviour);
            foreach (var col in model.GetComponentsInChildren<Collider>(true)) Object.DestroyImmediate(col);
            foreach (var lamp in model.GetComponentsInChildren<Light>(true)) Object.DestroyImmediate(lamp.gameObject);
            foreach (var camera in model.GetComponentsInChildren<Camera>(true)) Object.DestroyImmediate(camera.gameObject);
            Fit(model, 2.8f);
        }
        foreach (Transform child in root.GetComponentsInChildren<Transform>(true)) child.gameObject.layer = 11;
        foreach (var col in visual.GetComponentsInChildren<Collider>(true)) Object.DestroyImmediate(col);
        var collider = root.AddComponent<CapsuleCollider>(); collider.center = Vector3.up * 1.4f; collider.height = 2.8f; collider.radius = .6f;
        var unit = root.AddComponent<BattleUnit>(); unit.enemy = data; unit.enabled = false;
        var agent = root.GetComponent<NavMeshAgent>(); agent.enabled = false; agent.radius = .6f; agent.height = 2.8f; agent.obstacleAvoidanceType = ObstacleAvoidanceType.HighQualityObstacleAvoidance;
        if (data.kind == BattleEnemyKind.Skeleton) DungeonPolishSetup.FitSkeletonCollider(root);
        var encounter = root.AddComponent<EncounterEnemy>(); encounter.data = data; encounter.IsBoss = data.kind == BattleEnemyKind.Jevil;
        var highlight = root.AddComponent<ObjectHighlight>(); var fields = new SerializedObject(highlight); fields.FindProperty("outlineColor").colorValue = new Color(1, .12f, .14f); fields.ApplyModifiedPropertiesWithoutUndo();
        try { return PrefabUtility.SaveAsPrefabAsset(root, AssetsFolder + "/" + data.displayName + ".prefab"); } finally { Object.DestroyImmediate(root); }
    }
    static BattleLayout Layout(int index, DungeonSettings settings)
    {
        var root = new GameObject("Layout" + (index + 1)); root.transform.position = new Vector3(0, -1500, 0);
        var layout = root.AddComponent<BattleLayout>(); layout.surface = root.AddComponent<NavMeshSurface>(); layout.surface.collectObjects = CollectObjects.Children; layout.surface.useGeometry = NavMeshCollectGeometry.PhysicsColliders; layout.surface.layerMask = 1;
        var floor = new GameObject("Floor"); floor.transform.SetParent(root.transform, false); var collider = floor.AddComponent<BoxCollider>(); collider.center = Vector3.down * .3f; collider.size = new Vector3(66, .6f, 66);
        for (int x = 0; x < 12; x++) for (int z = 0; z < 12; z++)
        {
            var tile = Object.Instantiate(settings.floorTiles[(x * 7 + z * 3 + index) % settings.floorTiles.Length], floor.transform); tile.name = "Tile_" + x + "_" + z;
            tile.transform.localPosition = Vector3.zero; tile.transform.localRotation = Quaternion.identity;
            var bounds = Bounds(tile); tile.transform.localScale *= 5.5f / Mathf.Max(bounds.size.x, bounds.size.z);
            bounds = Bounds(tile); tile.transform.position += root.transform.position + new Vector3((x - 5.5f) * 5.5f, 0, (z - 5.5f) * 5.5f) - new Vector3(bounds.center.x, bounds.max.y, bounds.center.z);
            foreach (var col in tile.GetComponentsInChildren<Collider>()) col.enabled = false;
        }
        layout.partySpawns = new Transform[3]; layout.enemySpawns = new Transform[7];
        Vector3[][] party = { new[] { new Vector3(-7, 0, -10), new Vector3(0, 0, -12), new Vector3(7, 0, -10) }, new[] { new Vector3(-12, 0, -7), new Vector3(-13, 0, 0), new Vector3(-12, 0, 7) }, new[] { new Vector3(-10, 0, -8), new Vector3(0, 0, -13), new Vector3(10, 0, -8) } };
        for (int i = 0; i < 3; i++) layout.partySpawns[i] = Joint("PartySpawn" + i, root.transform, party[index][i] * 1.5f);
        for (int i = 0; i < 7; i++) layout.enemySpawns[i] = Joint("EnemySpawn" + i, root.transform, (index == 1 ? new Vector3(10 + i % 2 * 4, 0, (i - 3) * 4) : new Vector3((i - 3) * 4, 0, 8 + i % 2 * 5)) * 1.5f);
        layout.cover = new BattleCover[5];
        Vector3[][] cover = { new[] { new Vector3(-12, 0, 1), new Vector3(-4, 0, -3), new Vector3(4, 0, 2), new Vector3(12, 0, -2), new Vector3(0, 0, 14) }, new[] { new Vector3(-5, 0, -10), new Vector3(0, 0, -4), new Vector3(0, 0, 5), new Vector3(5, 0, 11), new Vector3(-9, 0, 13) }, new[] { new Vector3(-13, 0, -1), new Vector3(-5, 0, 3), new Vector3(5, 0, 3), new Vector3(13, 0, -1), new Vector3(0, 0, 12) } };
        for (int i = 0; i < 5; i++)
        {
            var wall = new GameObject("Cover" + i); wall.transform.SetParent(root.transform, false); wall.transform.localPosition = cover[index][i] * 1.5f;
            var stone = settings.wallMaterial != null ? settings.wallMaterial : Material("CoverStone", new Color(.27f, .31f, .4f));
            var shape = Part("StoneBottom", wall.transform, Vector3.up * .75f, new Vector3(3.2f, 1.5f, 1.3f), stone);
            var top = Part("StoneTop", wall.transform, Vector3.up * 2.25f, new Vector3(3.2f, 1.5f, 1.3f), stone);
            Object.DestroyImmediate(shape.GetComponent<Collider>()); Object.DestroyImmediate(top.GetComponent<Collider>());
            var box = wall.AddComponent<BoxCollider>(); box.center = Vector3.up * 1.5f; box.size = new Vector3(3.2f, 3, 1.3f);
            var c = wall.AddComponent<BattleCover>(); c.shape = shape.transform; c.box = box; c.topHalf = top;
            c.breakEffect = Load<GameObject>(AssetsFolder + "/CoverBreak.prefab"); layout.cover[i] = c;
        }
        root.SetActive(false); return layout;
    }

    static void BuildHUD(BattleHUD hud, CharacterBase[] characters)
    {
        var parent = hud.transform; hud.turn = Text("Turn", parent, 0, 478, 600, 70, "Your Turn", 42);
        hud.portraits = new Button[3]; hud.hpBars = new Image[3]; hud.spBars = new Image[3]; hud.characterValues = new TMP_Text[3];
        for (int i = 0; i < 3; i++)
        {
            var row = Box("Character" + i, parent, -740, -100 - i * 130, 380, 115, Blue);
            var button = row.gameObject.AddComponent<Button>(); button.targetGraphic = row.GetComponent<Image>(); hud.portraits[i] = button;

    static void AddAnimations(Animator animator)
    {
        if (animator == null) return;
        var asset = animator.runtimeAnimatorController;
        if (asset is AnimatorOverrideController overrides) asset = overrides.runtimeAnimatorController;
        if (!(asset is AnimatorController controller)) return;
        var machine = controller.layers[0].stateMachine;
        string[] names = { "BattleAttack", "BattleCast", "BattleRollBack", "BattleRollSide" };
        string[] paths = { "Combat/MeeleeAttack_OneHanded.fbx", "Combat/SpellCast.fbx", "Movement/RollBackward.fbx", "Movement/RollLeft.fbx" };
        for (int i = 0; i < names.Length; i++)
        {
            var state = machine.states.Select(s => s.state).FirstOrDefault(s => s.name == names[i]) ?? machine.AddState(names[i]);
            state.motion = AssetDatabase.LoadAllAssetsAtPath("Assets/Blink/Art/Animations/" + paths[i]).OfType<AnimationClip>().FirstOrDefault(c => !c.name.StartsWith("__"));
            if (state.transitions.Length == 0) { var exit = state.AddTransition(machine.states.Select(s => s.state).First(s => s.name == "Idle")); exit.hasExitTime = true; exit.exitTime = .9f; exit.duration = .1f; }
        }
        EditorUtility.SetDirty(controller);
    }

    static Sprite SpriteAsset(string path)
    {
        var sprite = Load<Sprite>(path);
        if (sprite != null) return sprite;
        var importer = AssetImporter.GetAtPath(path) as TextureImporter;
        if (importer == null) throw new FileNotFoundException("Missing battle image", path);
        importer.textureType = TextureImporterType.Sprite; importer.spriteImportMode = SpriteImportMode.Single;
        importer.SaveAndReimport(); return Load<Sprite>(path) ?? throw new InvalidOperationException("Sprite import failed: " + path);
    }
